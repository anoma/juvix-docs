module reference.language.records;

type T := construct : T;

type NewType :=
  mk@{
    f : T;
  };

type Pair (A B : Type) :=
  mk@{
    fst : A;
    snd : B;
  };

type EnumRecord :=
  | C1@{
      c1a : T;
      c1b : T;
    }
  | C2@{
      c2a : T;
      c2b : T;
    };

p1 : Pair T T :=
  Pair.mk@{
    fst := T.construct;
    snd := T.construct;
  };

p2 : Pair EnumRecord EnumRecord :=
  Pair.mk@{
    fst :=
      EnumRecord.C1@{
        c1a := T.construct;
        c1b := T.construct;
      };
    snd :=
      EnumRecord.C2@{
        c2a := T.construct;
        c2b := T.construct;
      };
  };

p1' : Pair T T :=
  Pair.mk@{
    fst := Pair.fst p1;
    snd := Pair.snd p1;
  };

module Open-Pair;
  open Pair;
  
  flipP : Pair T T := Pair.mk (snd p1) (fst p1);
end;

import Stdlib.Data.Nat open;

natPair : Pair Nat Nat :=
  Pair.mk@{
    fst := 1;
    snd := 2;
  };

updatedNatPair : Pair Nat Nat := natPair@Pair{fst := 2};

p3 : Pair Nat Nat :=
  let
    fst := 1;
    snd := 2;
  in Pair.mk@{
       fst := fst;
       snd := snd;
     };

p3' : Pair Nat Nat :=
  let
    fst := 1;
    snd := 2;
  in Pair.mk@{
       fst;
       snd;
     };

f : Pair Nat Nat -> Nat
  | Pair.mk@{fst := zero; snd} := snd
  | Pair.mk@{fst} := fst;
